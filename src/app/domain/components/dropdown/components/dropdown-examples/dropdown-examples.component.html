    // .ts
    export class DropdownExamplesComponent implements OnInit {
        keyText: string = 'name';
    
        data: any[] = [
            { name: 'Elephant', id: 1 },
            { name: 'Lion', id: 2 },
            { name: 'Tiger', id: 3 },
            { name: 'Bear', id: 4 },
            { name: 'Wolf', id: 5 },
            { name: 'Fox', id: 6 }
            { name: 'Dog', id: 7 },
            { name: 'Cat', id: 8 },
            { name: 'Turtle', id: 9 },
            { name: 'Dolphin', id: 10 }
        ];
    
        dataToShow = signal<any[]>([]);
    
        ngOnInit() {
            this.dataToShow.set(this.data);
        }
    
        search(value: string) {
            this.dataToShow.set([...this.data].filter((item) => 
                item[this.keyText].toLowerCase().includes(value.toLowerCase())));
        }
    }
